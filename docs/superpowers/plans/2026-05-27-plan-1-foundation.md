# Plan 1 — Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Postgres schema, the `drafts-mcp` MCP server, and a minimal Next.js web shell with a working approval queue — no agents, no platform APIs. After this ships, you can manually create a draft, move it through the entire state machine, click Approve, and see it logged as `posted`.

**Architecture:** One Postgres database with Drizzle ORM. `drafts-mcp` is a TypeScript MCP server (stdio JSON-RPC) exposing CRUD over drafts; the same package re-exports its Drizzle schema for the web shell. Next.js 15 (App Router) with shadcn/ui consumes that schema directly (server components hitting Postgres). No external services in Plan 1.

**Tech Stack:** TypeScript, pnpm workspaces, Postgres 16, Drizzle ORM + drizzle-kit, Next.js 15, shadcn/ui, Tailwind CSS, vitest, Playwright, @modelcontextprotocol/sdk, Docker Compose.

**Key boundaries:**

- `@paperclip/db` owns the schema; everything else imports from it. No raw SQL elsewhere.
- `@paperclip/drafts-mcp` is the *agent-facing* CRUD API. It enforces the state machine.
- `@paperclip/web` is the *human-facing* UI. For Plan 1, it talks to Postgres directly via Drizzle from server components and `/app/api/*` route handlers — same state-machine enforcement, same Drizzle types.
- Both consumers go through one small `stateMachine.ts` module to keep transition rules in one place. **DRY is enforced by sharing this module.**

```ts
    research: jsonb('research'),

    statusIdx: index('drafts_status_idx').on(t.status),

  }),

  'draft_transitions',
  {
    id: uuid('id').primaryKey().defaultRandom(),
    draftId: uuid('draft_id').notNull().references(() => drafts.id, { onDelete: 'cascade' }),
    fromStatus: text('from_status'),
    toStatus: text('to_status').notNull(),
```

```ts
import { DraftStatus } from '@paperclip/db';

const allowed: Record<DraftStatus, DraftStatus[]> = {
  [DraftStatus.Briefed]: [DraftStatus.Researched],
  [DraftStatus.Researched]: [DraftStatus.Drafted],
  [DraftStatus.Drafted]: [DraftStatus.EditorReview],
  [DraftStatus.EditorReview]: [DraftStatus.ApprovalQueue, DraftStatus.ReworkRequested, DraftStatus.EditorRejected],
  [DraftStatus.ReworkRequested]: [DraftStatus.Drafted],
  [DraftStatus.ApprovalQueue]: [DraftStatus.Approved, DraftStatus.Rejected],
  [DraftStatus.Approved]: [DraftStatus.Scheduled, DraftStatus.AwaitingPaste],
  [DraftStatus.Scheduled]: [DraftStatus.Posted, DraftStatus.PostFailed],
  [DraftStatus.AwaitingPaste]: [DraftStatus.Posted],
  [DraftStatus.Posted]: [DraftStatus.Measuring],
  [DraftStatus.Measuring]: [DraftStatus.Archived],
  [DraftStatus.PostFailed]: [DraftStatus.Approved], // human can retry
  [DraftStatus.Rejected]: [], // terminal
  [DraftStatus.EditorRejected]: [DraftStatus.ApprovalQueue, DraftStatus.Rejected], // human override
  [DraftStatus.Archived]: [], // terminal
};

export function canTransition(from: DraftStatus, to: DraftStatus): boolean {
  return allowed[from]?.includes(to) ?? false;
}

export function assertTransition(from: DraftStatus, to: DraftStatus): void {
  if (!canTransition(from, to)) {
    throw new Error(`invalid transition: ${from} -> ${to}`);
  }
}

export const allowedTransitions = allowed;
```

```ts
import { db, drafts, draftTransitions, DraftStatus } from '@paperclip/db';
import { eq } from 'drizzle-orm';
import { assertTransition } from '../stateMachine.js';

export const REWORK_CAP = 2;

export interface TransitionStatusInput {
  draftId: string;
  to: DraftStatus;
  actor: string;
  reason?: string;
}

export async function transitionStatus(input: TransitionStatusInput) {
  return db.transaction(async (tx) => {
    const [current] = await tx.select().from(drafts).where(eq(drafts.id, input.draftId));
    if (!current) throw new Error(`draft not found: ${input.draftId}`);

    assertTransition(current.status as DraftStatus, input.to);

    let newReworkCount = current.reworkCount;
    if (input.to === DraftStatus.ReworkRequested) {
      if (current.reworkCount >= REWORK_CAP) {
        throw new Error(
          `rework cap exceeded (${REWORK_CAP}); editor must transition to editor_rejected instead`,
        );
      }
      newReworkCount = current.reworkCount + 1;
    }

    const [updated] = await tx
      .update(drafts)
      .set({
        status: input.to,
        reworkCount: newReworkCount,
        updatedAt: new Date(),
      })
      .where(eq(drafts.id, input.draftId))
      .returning();

    await tx.insert(draftTransitions).values({
      draftId: input.draftId,
      fromStatus: current.status,
      toStatus: input.to,
      actor: input.actor,
      reason: input.reason,
    });

    return updated!;
  });
}
```

```ts
import { Server } from '@modelcontextprotocol/sdk/server/index.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import { CallToolRequestSchema, ListToolsRequestSchema } from '@modelcontextprotocol/sdk/types.js';
import { z } from 'zod';
import { createDraft } from './tools/createDraft.js';
import { updateDraft } from './tools/updateDraft.js';
import { getDraft } from './tools/getDraft.js';
import { listByStatus } from './tools/listByStatus.js';
import { transitionStatus } from './tools/transitionStatus.js';
import { DraftStatus, Platform } from '@paperclip/db';

const createDraftSchema = z.object({
  platform: z.enum([Platform.X, Platform.LinkedIn]),
  status: z.string().optional(),
  themeId: z.string().uuid().optional(),
  parentBriefId: z.string().uuid().optional(),
  body: z.string().optional(),
  hookTag: z.string().optional(),
});

const updateDraftSchema = z.object({
  id: z.string().uuid(),
  body: z.string().optional(),
  hookTag: z.string().optional(),
  editorNotes: z.unknown().optional(),
  scheduledFor: z.string().datetime().optional(),
});

const transitionSchema = z.object({
  draftId: z.string().uuid(),
  to: z.string(),
  actor: z.string(),
  reason: z.string().optional(),
});

const listSchema = z.object({
  status: z.string(),
  platform: z.enum([Platform.X, Platform.LinkedIn]).optional(),
  limit: z.number().int().min(1).max(500).optional(),
});

const getSchema = z.object({ id: z.string().uuid() });

const tools = [
  { name: 'createDraft', description: 'Create a new draft row', schema: createDraftSchema },
  { name: 'updateDraft', description: 'Update non-status fields', schema: updateDraftSchema },
  { name: 'transitionStatus', description: 'Transition a draft status (records audit row)', schema: transitionSchema },
  { name: 'listByStatus', description: 'List drafts by status (and optional platform)', schema: listSchema },
  { name: 'getDraft', description: 'Get one draft by id', schema: getSchema },
] as const;

const server = new Server(
  { name: 'drafts-mcp', version: '0.0.0' },
  { capabilities: { tools: {} } },
);

server.setRequestHandler(ListToolsRequestSchema, async () => ({
  tools: tools.map((t) => ({
    name: t.name,
    description: t.description,
    inputSchema: {
      type: 'object',
      properties: {},
      additionalProperties: true,
    },
  })),
}));

server.setRequestHandler(CallToolRequestSchema, async (req) => {
  const { name, arguments: args } = req.params;
  try {
    let result: unknown;
    switch (name) {
      case 'createDraft': {
        const v = createDraftSchema.parse(args);
        result = await createDraft({ ...v, status: v.status as DraftStatus | undefined });
        break;
      }
      case 'updateDraft': {
        const v = updateDraftSchema.parse(args);
        result = await updateDraft({
          ...v,
          scheduledFor: v.scheduledFor ? new Date(v.scheduledFor) : undefined,
        });
        break;
      }
      case 'transitionStatus': {
        const v = transitionSchema.parse(args);
        result = await transitionStatus({ ...v, to: v.to as DraftStatus });
        break;
      }
      case 'listByStatus': {
        const v = listSchema.parse(args);
        result = await listByStatus({ ...v, status: v.status as DraftStatus });
        break;
      }
      case 'getDraft': {
        const v = getSchema.parse(args);
        result = await getDraft(v);
        break;
      }
      default:
        throw new Error(`unknown tool: ${name}`);
    }
    return { content: [{ type: 'text', text: JSON.stringify(result, null, 2) }] };
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    return { content: [{ type: 'text', text: `ERROR: ${message}` }], isError: true };
  }
});

const transport = new StdioServerTransport();
await server.connect(transport);
```

```ts
    const [draft] = await tx.select().from(drafts).where(eq(drafts.id, id));
    if (!draft) throw new Error('not_found');
    const to = body.data.to as DraftStatus;
    const allowedFrom = allowed[draft.status] ?? [];
    if (!allowedFrom.includes(to)) throw new Error(`invalid_transition:${draft.status}->${to}`);
    const [updated] = await tx
      .update(drafts)
      .set({ status: to, updatedAt: new Date() })
      .where(eq(drafts.id, id))
      .returning();
    await tx.insert(draftTransitions).values({
      draftId: id,
      fromStatus: draft.status,
      toStatus: to,
      actor: body.data.actor,
      reason: body.data.reason,
    });
    return updated!;
  }).catch((err: Error) => ({ error: err.message }));

  if ('error' in result) {
    return NextResponse.json(result, { status: result.error === 'not_found' ? 404 : 409 });
  }
  return NextResponse.json(result);
}
```

> **Note:** The `allowed` map is duplicated from `drafts-mcp/src/stateMachine.ts`. Task 11 extracts it into a shared module to eliminate this duplication.

```ts
  [DraftStatus.Rejected]: [],
  [DraftStatus.EditorRejected]: [DraftStatus.ApprovalQueue, DraftStatus.Rejected],
  [DraftStatus.Archived]: [],
};

export function canTransition(from: DraftStatus, to: DraftStatus): boolean {
  return allowed[from]?.includes(to) ?? false;
}

export function assertTransition(from: DraftStatus, to: DraftStatus): void {
  if (!canTransition(from, to)) throw new Error(`invalid transition: ${from} -> ${to}`);
}

export const allowedTransitions = allowed;
```

```tsx
import { db, drafts, draftTransitions } from '@paperclip/db';
```
